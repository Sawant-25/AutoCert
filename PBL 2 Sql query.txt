CREATE TABLE sponsor_logos (
  id SERIAL PRIMARY KEY,
  request_id INTEGER REFERENCES requests(id) ON DELETE CASCADE,
  logo TEXT
);
CREATE TABLE assets (
  id SERIAL PRIMARY KEY,
  request_id INTEGER REFERENCES requests(id) ON DELETE CASCADE UNIQUE,

  college_logo TEXT,
  event_logo TEXT,
  faculty_signature TEXT,

  hod_signature TEXT,
  director_signature TEXT
);
